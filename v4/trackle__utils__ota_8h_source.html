71esp_err_t _http_event_handler(esp_http_client_event_t *evt)
72{
73 switch (evt->event_id)
74 {
75 case HTTP_EVENT_ERROR:
76 ESP_LOGI(OTA_TAG, "HTTP_EVENT_ERROR");
77 break;
78 case HTTP_EVENT_ON_CONNECTED:
79 ESP_LOGI(OTA_TAG, "HTTP_EVENT_ON_CONNECTED");
80 break;
81 case HTTP_EVENT_HEADER_SENT:
82 ESP_LOGI(OTA_TAG, "HTTP_EVENT_HEADER_SENT");
83 break;
84 case HTTP_EVENT_ON_HEADER:
85 ESP_LOGI(OTA_TAG, "HTTP_EVENT_ON_HEADER, key=%s, value=%s", evt->header_key, evt->header_value);
86 break;
87 case HTTP_EVENT_ON_DATA:
88 ESP_LOGI(OTA_TAG, "HTTP_EVENT_ON_DATA, len=%d", evt->data_len);
89 current_ota_data.actual_crc32_ota = crc32_le(current_ota_data.actual_crc32_ota, evt->data, evt->data_len);
90 // TODO update percentage
91 break;
92 case HTTP_EVENT_ON_FINISH:
93 ESP_LOGI(OTA_TAG, "HTTP_EVENT_ON_FINISH");
94 break;
95 case HTTP_EVENT_DISCONNECTED:
96 ESP_LOGI(OTA_TAG, "HTTP_EVENT_DISCONNECTED");
97 break;
98 default:
99 ESP_LOGI(OTA_TAG, "OTHER HTTP_EVENT");
100 }
101 return ESP_OK;
102}

120void execute_ota_task(void *pvParameter)
121{
122 ESP_LOGI(OTA_TAG, "Starting OTA %s", current_ota_data.url);
123 current_ota_data.start_timestamp = getMillis();
124
125 xEventGroupSetBits(s_wifi_event_group, OTA_UPDATING);
126
127 esp_http_client_config_t config = {
128 .url = current_ota_data.url,
129 .event_handler = _http_event_handler,
130 .buffer_size = 1024,
131 };
132
133 esp_https_ota_config_t ota_config = {
134 .http_config = &config,
135 };
136
137 esp_https_ota_handle_t https_ota_handle = NULL;
138 esp_err_t err = esp_https_ota_begin(&ota_config, &https_ota_handle);
139
140 if (err == ESP_ERR_INVALID_ARG || err == ESP_ERR_OTA_PARTITION_CONFLICT || err == ESP_ERR_OTA_SELECT_INFO_INVALID || err == ESP_ERR_INVALID_SIZE || err == ESP_ERR_OTA_ROLLBACK_INVALID_STATE || err == ESP_ERR_NOT_FOUND)
141 {
142 sendOtaMessage(OTA_MSG_DONE, OTA_ERR_PARTITION);
143 }
144 else if (err == ESP_ERR_NO_MEM)
145 {
146 sendOtaMessage(OTA_MSG_DONE, OTA_ERR_MEMORY);
147 }
148 else if (err != ESP_OK)
149 {
150 sendOtaMessage(OTA_MSG_DONE, OTA_ERR_GENERIC);
151 }
152 else
153 {
154 while (1)
155 {
156 err = esp_https_ota_perform(https_ota_handle);
157 if (err != ESP_ERR_HTTPS_OTA_IN_PROGRESS)
158 {
159 break;
160 }
161 }
162
163 if (err != ESP_OK || https_ota_handle == NULL || esp_https_ota_is_complete_data_received(https_ota_handle) != true)
164 {
165 ESP_LOGE(OTA_TAG, "Complete data was not received.");
166 sendOtaMessage(OTA_MSG_DONE, OTA_ERR_INCOMPLETE);
167 }
168 else
169 {
170 // check crc
171 ESP_LOGI(OTA_TAG, "current_ota_data.actual_crc32_ota %" PRIu32, current_ota_data.actual_crc32_ota);
172 ESP_LOGI(OTA_TAG, "current_ota_data.firmware_crc32_ota %" PRIu32, current_ota_data.firmware_crc32_ota);
173
174 if (current_ota_data.firmware_crc32_ota == 0 || current_ota_data.firmware_crc32_ota == current_ota_data.actual_crc32_ota)
175 {
176 err = esp_https_ota_finish(https_ota_handle);
177 if (err == ESP_OK)
178 {
179 ESP_LOGI(OTA_TAG, "OTA completed, now restarting....");
180 sendOtaMessage(OTA_MSG_DONE, OTA_ERR_OK);
181 vTaskDelay(1000 / portTICK_PERIOD_MS);
182 esp_restart();
183 }
184 else
185 {
186 sendOtaMessage(OTA_MSG_DONE, OTA_ERR_COMPLETING);
187 }
188 }
189 else
190 {
191 sendOtaMessage(OTA_MSG_DONE, OTA_ERR_VALIDATE_FAILED);
192 }
193 }
194 }
195
196 ESP_LOGE(OTA_TAG, "ESP_HTTPS_OTA upgrade failed");
197 vTaskDelay(1000 / portTICK_PERIOD_MS);
198 xEventGroupClearBits(s_wifi_event_group, OTA_UPDATING); // stop updating
199 current_ota_data.start_timestamp = 0;
200 esp_https_ota_abort(https_ota_handle);
201 vTaskDelete(xOtaTaskHandle);
202}

210int firmware_ota_url(const char *url, uint32_t crc)
211{
212 // check if reset start_timestamp
213 EventBits_t bits = xEventGroupGetBits(s_wifi_event_group);
214 if (bits & OTA_UPDATING)
215 {
216 // delete task and reset counter, then start ota
217 if (current_ota_data.start_timestamp > 0 && getMillis() - current_ota_data.start_timestamp >= OTA_TIMEOUT)
218 {
219 current_ota_data.start_timestamp = 0;
220 if (xOtaTaskHandle != NULL)
221 {
222 vTaskDelete(xOtaTaskHandle);
223 }
224 xEventGroupClearBits(s_wifi_event_group, OTA_UPDATING);
225 }
226 else // return error
227 {
228 return OTA_ERR_ALREADY_RUNNING;
229 }
230 }
231
232 ESP_LOGI(OTA_TAG, "ota update callback, url: %s, crc %" PRIu32, url, crc);
233 strcpy(current_ota_data.url, url);
234 current_ota_data.firmware_crc32_ota = crc;
235 current_ota_data.actual_crc32_ota = 0;
236 xTaskCreate(&execute_ota_task, "execute_ota_task", 8192, NULL, 5, &xOtaTaskHandle);
237 return OTA_ERR_OK;
238}
