48esp_err_t wifi_is_provisioned()
49{
50 /* Get Wi-Fi Station configuration */

67static void event_handler(void *arg, esp_event_base_t event_base,
68 int32_t event_id, void *event_data)
69{
70 EventBits_t bits = xEventGroupGetBits(s_wifi_event_group);
71
72 if (event_base == WIFI_EVENT && event_id == WIFI_EVENT_STA_START)
73 {
74 ESP_LOGI(WIFI_TAG, "Wifi started.....");
75
76 wifi_mode_t currentMode;
77 esp_wifi_get_mode(&currentMode);
78 if (currentMode == WIFI_MODE_STA)
79 {
80 ESP_LOGI(WIFI_TAG, "Connecting to the AP");
81 xEventGroupSetBits(s_wifi_event_group, WIFI_TO_CONNECT_BIT); // connettiti
82 timeout_connect_wifi = getMillis();
83 }
84 else if (currentMode == WIFI_MODE_APSTA)
85 {
86 ESP_LOGI(WIFI_TAG, "APMode, not connecting....");
87 xEventGroupClearBits(s_wifi_event_group, WIFI_TO_CONNECT_BIT); // non cercare di riconnetterti
88 }
89 }
90 else if (event_base == WIFI_EVENT && event_id == WIFI_EVENT_STA_DISCONNECTED)
91 {
92 wifi_event_sta_disconnected_t *event = (wifi_event_sta_disconnected_t *)event_data;
93 ESP_LOGW(WIFI_TAG, "Wifi disconnection event: %d...", event->reason);
94
95 if (bits & NETWORK_CONNECTED_BIT)
96 {
97 trackleDiagnosticNetwork(trackle_s, NETWORK_DISCONNECTS, 1);
98
99 // reset connections attemps for new cloud session
100 trackleDiagnosticNetwork(trackle_s, NETWORK_CONNECTION_ATTEMPTS, 0);
101 }
102
103 timeout_connect_wifi = getMillis() + CHECK_WIFI_TIMEOUT;
104 xEventGroupClearBits(s_wifi_event_group, NETWORK_CONNECTED_BIT);
105 }
106 else if (event_base == IP_EVENT && event_id == IP_EVENT_STA_GOT_IP)
107 {
108 ip_event_got_ip_t *event = (ip_event_got_ip_t *)event_data;
109 ESP_LOGW(WIFI_TAG, "Got ip:" IPSTR, IP2STR(&event->ip_info.ip));
110 xEventGroupSetBits(s_wifi_event_group, NETWORK_CONNECTED_BIT);
111
112 // diagnostic
113 esp_wifi_sta_get_ap_info(&ap);
114 trackleDiagnosticNetwork(trackle_s, NETWORK_IPV4_ADDRESS, (int32_t)(event->ip_info.ip.addr));
115 trackleDiagnosticNetwork(trackle_s, NETWORK_IPV4_GATEWAY, (int32_t)event->ip_info.gw.addr);
116 trackleDiagnosticNetwork(trackle_s, NETWORK_RSSI, ap.rssi);
117 trackleDiagnosticNetwork(trackle_s, NETWORK_SIGNAL_STRENGTH, rssiToPercentage(ap.rssi));
118 }
119}

125void wifi_init()
126{
127 ESP_LOGI(WIFI_TAG, "wifi_init....");
128
129 // init event group
130 s_wifi_event_group = xEventGroupCreate();
131 ESP_ERROR_CHECK(esp_netif_init());
132 ESP_ERROR_CHECK(esp_event_loop_create_default());
133
134 // Register our event handler for Wi-Fi, IP and Provisioning related events
135 ESP_ERROR_CHECK(esp_event_handler_register(WIFI_EVENT, ESP_EVENT_ANY_ID, &event_handler, NULL));
136 ESP_ERROR_CHECK(esp_event_handler_register(IP_EVENT, IP_EVENT_STA_GOT_IP, &event_handler, NULL));
137
138 sta_netif = esp_netif_create_default_wifi_sta();
139 wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
140 ESP_ERROR_CHECK(esp_wifi_init(&cfg));
141 esp_wifi_set_storage(WIFI_STORAGE_FLASH);
142}

158void trackle_utils_wifi_loop()
159{
160 EventBits_t bits = xEventGroupGetBits(s_wifi_event_group);
161 uint32_t loop_millis = getMillis();
162
163 // check connessione wifi
164 if (timeout_connect_wifi > 0 && timeout_connect_wifi < loop_millis)
165 {
166 timeout_connect_wifi = 0;
167 if ((bits & WIFI_TO_CONNECT_BIT))
168 {
169 ESP_LOGI(WIFI_TAG, "Trying to connect to the AP...");
170 esp_wifi_connect();
171
172 trackleDiagnosticNetwork(trackle_s, NETWORK_CONNECTION_ATTEMPTS, 1);
173 }
174 }
175
176 // updating diagnostic
177 if (getMillis() - utility_check_diagnostic_millis >= UTILITY_DIAGNOSTIC_TIME)
178 {
179 utility_check_diagnostic_millis = getMillis();
180 if (bits & NETWORK_CONNECTED_BIT)
181 {
182 esp_wifi_sta_get_ap_info(&ap);
183 trackleDiagnosticNetwork(trackle_s, NETWORK_RSSI, ap.rssi);
184 trackleDiagnosticNetwork(trackle_s, NETWORK_SIGNAL_STRENGTH, rssiToPercentage(ap.rssi));
185 }
186 }
187}
