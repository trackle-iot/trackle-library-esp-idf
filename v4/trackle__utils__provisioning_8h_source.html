166// Universal read line function
167static inline int udc_read_line(char *buffer, size_t max_length, uint32_t timeout_ms)
168{
169 int pos = 0;
170 char c;
171 TickType_t start_time = xTaskGetTickCount();
172 TickType_t timeout_ticks = pdMS_TO_TICKS(timeout_ms);
173
174 while (pos < max_length - 1)
175 {
176 TickType_t now = xTaskGetTickCount();
177
178 // Check timeout first
179 if (now - start_time > timeout_ticks)
180 {
181 ESP_LOGW(UDC_TAG, "Timeout occurred");
182 buffer[pos] = '\0';
183 return -1; // Return -1 to indicate timeout
184 }
185
186#if UDC_USE_USB_SERIAL_JTAG
187 int len = usb_serial_jtag_read_bytes(&c, 1, pdMS_TO_TICKS(100));
188#else
189 int len = uart_read_bytes(UDC_UART_NUM, &c, 1, pdMS_TO_TICKS(100));
190#endif
191
192 if (len > 0)
193 {
194 if (c == '\n' || c == '\r')
195 {
196 break;
197 }
198 else if (c == '\b' || c == 127)
199 {
200 if (pos > 0)
201 {
202 pos--;
203 udc_send("\b \b");
204 }
205 }
206 else if (c >= 32 && c <= 126)
207 {
208 buffer[pos++] = c;
209#if UDC_USE_USB_SERIAL_JTAG
210 usb_serial_jtag_write_bytes(&c, 1, portMAX_DELAY);
211#else
212 uart_write_bytes(UDC_UART_NUM, &c, 1);
213#endif
214 }
215 }
216 else
217 {
218 // Brief delay to prevent watchdog timeout and reduce CPU usage
219 vTaskDelay(pdMS_TO_TICKS(10));
220 }
221 }
222
223 buffer[pos] = '\0';
224 udc_send("\r\n");
225 return pos;
226}

228// Convert string to boolean
229static inline bool udc_to_bool(const char *input, bool *result)
230{
231 char lower[32];
232 strncpy(lower, input, sizeof(lower) - 1);
233 lower[sizeof(lower) - 1] = '\0';
234
235 for (int i = 0; lower[i]; i++)
236 {
237 lower[i] = tolower(lower[i]);
238 }
239
240 if (strcmp(lower, "true") == 0 || strcmp(lower, "yes") == 0 ||
241 strcmp(lower, "y") == 0 || strcmp(lower, "1") == 0 ||
242 strcmp(lower, "on") == 0 || strcmp(lower, "enable") == 0)
243 {
244 *result = true;
245 return true;
246 }
247
248 if (strcmp(lower, "false") == 0 || strcmp(lower, "no") == 0 ||
249 strcmp(lower, "n") == 0 || strcmp(lower, "0") == 0 ||
250 strcmp(lower, "off") == 0 || strcmp(lower, "disable") == 0)
251 {
252 *result = false;
253 return true;
254 }
255
256 return false;
257}

259// Convert hex string to buffer
260static inline bool udc_to_hex(const char *hex_string, uint8_t *buffer, size_t *length)
261{
262 size_t str_len = strlen(hex_string);
263 if (str_len % 2 != 0)
264 return false;
265
266 *length = str_len / 2;
267 for (size_t i = 0; i < *length; i++)
268 {
269 char hex_byte[3] = {hex_string[i * 2], hex_string[i * 2 + 1], '\0'};
270 for (int j = 0; j < 2; j++)
271 {
272 if (!isxdigit((unsigned char)hex_byte[j]))
273 return false;
274 }
275 buffer[i] = (uint8_t)strtol(hex_byte, NULL, 16);
276 }
277 return true;
278}

280// Validate and convert input
281static inline bool udc_validate(const char *input, udc_data_type_t type, udc_collected_data_t *result)
282{
283 switch (type)
284 {
285 case UDC_TYPE_STRING:
286 if (strlen(input) == 0)
287 return false;
288 strcpy(result->string_value, input);
289 return true;
290
291 case UDC_TYPE_HEX:
292 return udc_to_hex(input, result->hex_buffer, &result->hex_length);
293
294 case UDC_TYPE_INT:
295 if (strlen(input) == 0)
296 return false;
297 result->int_value = atoi(input);
298 return true;
299
300 case UDC_TYPE_FLOAT:
301 if (strlen(input) == 0)
302 return false;
303 result->float_value = atof(input);
304 return true;
305
306 case UDC_TYPE_BOOL:
307 if (strlen(input) == 0)
308 return false;
309 return udc_to_bool(input, &result->bool_value);
310
311 default:
312 return false;
313 }

316// Collect single data item
317static inline udc_result_t udc_collect_single(const udc_data_request_t *request, udc_collected_data_t *result)
318{
319 char input[UDC_MAX_INPUT_LENGTH];
320 char prompt[512];
321 int retry_count = 0;
322 const int max_retries = 3;
323
324 while (retry_count < max_retries)
325 {
326 // Brief delay to prevent watchdog timeout
327 vTaskDelay(pdMS_TO_TICKS(10));
328
329 const char *hint = "";
330 if (request->type == UDC_TYPE_BOOL)
331 hint = " (true/false, 1/0)";
332 if (request->type == UDC_TYPE_HEX)
333 hint = " (hex)";
334
335 snprintf(prompt, sizeof(prompt), "\r\n%s%s%s: ",
336 request->prompt,
337 request->required ? " (required)" : " (optional)",
338 hint);
339
340 udc_send(prompt);
341 int len = udc_read_line(input, sizeof(input), 30000); // 30s timeout per input
342
343 // Check for timeout
344 if (len == -1)
345 {
346 retry_count++;
347 if (retry_count >= max_retries)
348 {
349 udc_send("Timeout occurred. Collection failed.\r\n");
350 ESP_LOGE(UDC_TAG, "Timeout for field: %s", request->key);
351 return UDC_FAILED;
352 }
353 else
354 {
355 udc_send("Timeout. Please try again.\r\n");
356 continue;
357 }
358 }
359
360 // If optional and empty, accept
361 if (!request->required && len == 0)
362 {
363 ESP_LOGI(UDC_TAG, "Skipped: %s", request->key);
364 return UDC_SUCCESS;
365 }
366
367 // If required and empty, reject
368 if (request->required && len == 0)
369 {
370 udc_send("This field is required. Please enter a value.\r\n");
371 continue; // Don't increment retry_count for empty required fields
372 }
373
374 // Validate input
375 if (udc_validate(input, request->type, result))
376 {
377 ESP_LOGI(UDC_TAG, "Got: %s", request->key);
378 return UDC_SUCCESS;
379 }
380
381 // Input validation failed
382 udc_send("Invalid input. Try again.\r\n");
383 if (request->type == UDC_TYPE_HEX)
384 {
385 udc_send("Format: A1B2C3\r\n");
386 }
387 else if (request->type == UDC_TYPE_BOOL)
388 {
389 udc_send("Use: true/false, yes/no, 1/0\r\n");
390 }
391
392 // Only increment retry for validation failures, not empty required fields
393 retry_count++;
394 }
395
396 ESP_LOGE(UDC_TAG, "Max retries reached for field: %s", request->key);
397 return UDC_FAILED;
398}

445// Collection task function
446static inline void udc_collection_task(void *pvParameters)
447{
448 udc_task_params_t *params = (udc_task_params_t *)pvParameters;
449
450 if (!params || !params->requests || !params->results)
451 {
452 ESP_LOGE(UDC_TAG, "Invalid task parameters");
453 if (params)
454 {
455 if (params->result_code)
456 *params->result_code = UDC_INVALID_PARAMS;
457 if (params->done_flag)
458 *params->done_flag = true;
459 free(params);
460 }
461 vTaskDelete(NULL);
462 return;
463 }
464
465 ESP_LOGI(UDC_TAG, "Starting collection task with %zu requests via %s", params->count, UDC_CONNECTION_TYPE);
466
467 if (udc_init_uart() != ESP_OK)
468 {
469 ESP_LOGE(UDC_TAG, "%s init failed", UDC_CONNECTION_TYPE);
470 *params->result_code = UDC_FAILED;
471 *params->done_flag = true;
472 free(params);
473 vTaskDelete(NULL);
474 return;
475 }
476
477 vTaskDelay(pdMS_TO_TICKS(500));
478
479 udc_send("\r\n=== DATA COLLECTION ===\r\n");
480 char connection_info[128];
481 snprintf(connection_info, sizeof(connection_info), "Connection: %s\r\n", UDC_CONNECTION_TYPE);
482 udc_send(connection_info);
483 udc_send("Press CTRL+C to cancel\r\n\r\n");
484
485 udc_result_t final_result = UDC_SUCCESS;
486
487 for (size_t i = 0; i < params->count; i++)
488 {
489 ESP_LOGI(UDC_TAG, "Collecting %zu/%zu: %s", i + 1, params->count, params->requests[i].key);
490
491 if (udc_collect_single(&params->requests[i], &params->results[i]) != UDC_SUCCESS)
492 {
493 udc_send("Collection failed\r\n");
494 final_result = UDC_FAILED;
495 break;
496 }
497 }
498
499 if (final_result == UDC_SUCCESS)
500 {
501 udc_send("\r\nCollection complete!\r\n");
502 udc_show_summary(params->requests, params->count, params->results);
503 ESP_LOGI(UDC_TAG, "Collection completed successfully");
504 }
505
506 *params->result_code = final_result;
507 *params->done_flag = true;
508
509 ESP_LOGI(UDC_TAG, "Collection task finished");
510 free(params);
511 vTaskDelete(NULL);
512}

550static inline esp_err_t udc_collect_async(const udc_data_request_t *requests,
551 size_t count,
552 udc_collected_data_t *results,
553 volatile bool *done_flag,
554 volatile udc_result_t *result_code,
555 uint32_t stack_size,
556 UBaseType_t priority)
557{
558 if (!requests || !results || !done_flag || !result_code || count == 0)
559 {
560 return ESP_ERR_INVALID_ARG;
561 }
562
563 // Initialize flags
564 *done_flag = false;
565 *result_code = UDC_SUCCESS;
566
567 // Allocate task parameters (freed by task)
568 udc_task_params_t *params = malloc(sizeof(udc_task_params_t));
569 if (!params)
570 {
571 return ESP_ERR_NO_MEM;
572 }
573
574 params->requests = requests;
575 params->count = count;
576 params->results = results;
577 params->done_flag = done_flag;
578 params->result_code = result_code;
579
580 // Create task
581 TaskHandle_t task_handle;
582 BaseType_t task_result = xTaskCreate(udc_collection_task, "uart_collector",
583 stack_size, params, priority, &task_handle);
584
585 if (task_result != pdPASS)
586 {
587 ESP_LOGE(UDC_TAG, "Failed to create collection task");
588 free(params);
589 return ESP_FAIL;
590 }
591
592 ESP_LOGI(UDC_TAG, "Collection task created using %s", UDC_CONNECTION_TYPE);
593 return ESP_OK;
594}

626static inline esp_err_t udc_write_to_factory(const udc_data_request_t *requests, const udc_collected_data_t *results)
627{
628#define FACTORY_PARTITION "factory_data"
629#define FACTORY_NAMESPACE "device"
630
631 if (!requests || !results)
632 {
633 ESP_LOGE(UDC_TAG, "Invalid parameters for factory write");
634 return ESP_ERR_INVALID_ARG;
635 }
636
637 // Open factory_data partition in read-write mode
638 nvs_handle_t factory_handle;
639 esp_err_t err = nvs_open_from_partition(FACTORY_PARTITION, FACTORY_NAMESPACE, NVS_READWRITE, &factory_handle);
640 if (err != ESP_OK)
641 {
642 ESP_LOGE(UDC_TAG, "Failed to open factory partition: %s", esp_err_to_name(err));
643 return err;
644 }
645
646 ESP_LOGI(UDC_TAG, "Writing collected data to factory_data partition");
647
648 // Write all collected data
649 size_t write_count = 0;
650 for (size_t i = 0; requests[i].prompt != NULL; i++)
651 {
652 const char *key = requests[i].key;
653 const udc_collected_data_t *data = &results[i];
654
655 switch (requests[i].type)
656 {
657 case UDC_TYPE_STRING:
658 if (strlen(data->string_value) > 0)
659 {
660 err = nvs_set_str(factory_handle, key, data->string_value);
661 if (err == ESP_OK)
662 {
663 ESP_LOGI(UDC_TAG, "Written string '%s': %s", key, data->string_value);
664 write_count++;
665 }
666 else
667 {
668 ESP_LOGE(UDC_TAG, "Failed to write string '%s': %s", key, esp_err_to_name(err));
669 }
670 }
671 break;
672
673 case UDC_TYPE_HEX:
674 if (data->hex_length > 0)
675 {
676 err = nvs_set_blob(factory_handle, key, data->hex_buffer, data->hex_length);
677 if (err == ESP_OK)
678 {
679 ESP_LOGI(UDC_TAG, "Written hex blob '%s': %zu bytes", key, data->hex_length);
680 ESP_LOG_BUFFER_HEX(UDC_TAG, data->hex_buffer, data->hex_length);
681 write_count++;
682 }
683 else
684 {
685 ESP_LOGE(UDC_TAG, "Failed to write hex blob '%s': %s", key, esp_err_to_name(err));
686 }
687 }
688 break;
689
690 case UDC_TYPE_INT:
691 err = nvs_set_i32(factory_handle, key, data->int_value);
692 if (err == ESP_OK)
693 {
694 ESP_LOGI(UDC_TAG, "Written int '%s': %d", key, data->int_value);
695 write_count++;
696 }
697 else
698 {
699 ESP_LOGE(UDC_TAG, "Failed to write int '%s': %s", key, esp_err_to_name(err));
700 }
701 break;
702
703 case UDC_TYPE_FLOAT:
704 // Store float as string to avoid precision issues
705 char float_str[32];
706 snprintf(float_str, sizeof(float_str), "%.6f", data->float_value);
707 err = nvs_set_str(factory_handle, key, float_str);
708 if (err == ESP_OK)
709 {
710 ESP_LOGI(UDC_TAG, "Written float '%s': %s", key, float_str);
711 write_count++;
712 }
713 else
714 {
715 ESP_LOGE(UDC_TAG, "Failed to write float '%s': %s", key, esp_err_to_name(err));
716 }
717 break;
718
719 case UDC_TYPE_BOOL:
720 err = nvs_set_u8(factory_handle, key, data->bool_value ? 1 : 0);
721 if (err == ESP_OK)
722 {
723 ESP_LOGI(UDC_TAG, "Written bool '%s': %s", key, data->bool_value ? "true" : "false");
724 write_count++;
725 }
726 else
727 {
728 ESP_LOGE(UDC_TAG, "Failed to write bool '%s': %s", key, esp_err_to_name(err));
729 }
730 break;
731
732 default:
733 ESP_LOGW(UDC_TAG, "Unknown data type for key '%s', skipping", key);
734 break;
735 }
736
737 // Stop on first write error (optional - you can continue if preferred)
738 if (err != ESP_OK)
739 {
740 ESP_LOGE(UDC_TAG, "Stopping writes due to error");
741 break;
742 }
743 }
744
745 // Commit all changes
746 if (err == ESP_OK)
747 {
748 err = nvs_commit(factory_handle);
749 if (err == ESP_OK)
750 {
751 ESP_LOGI(UDC_TAG, "Successfully wrote %zu values to factory_data partition", write_count);
752 }
753 else
754 {
755 ESP_LOGE(UDC_TAG, "Failed to commit factory data: %s", esp_err_to_name(err));
756 }
757 }
758
759 // Close handle
760 nvs_close(factory_handle);
761
762 return err;
763}
